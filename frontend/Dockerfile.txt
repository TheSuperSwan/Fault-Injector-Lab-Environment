FROM registry.access.redhat.com/ubi10/nginx-126:10.0

# Install tools needed for fault-injection scripts
# - bash: scripts use /usr/bin/env bash
# - curl: scripts call the backend fault API via localhost (nginx proxy)
RUN microdnf install -y bash curl && microdnf clean all

RUN rm -rf /opt/app-root/src/*

COPY frontend/index.html /opt/app-root/src/index.html

COPY frontend/nginx.conf /etc/nginx/nginx.conf

COPY scripts/ /usr/local/bin/

RUN chmod +x /usr/local/bin/fi-*.sh

EXPOSE 8080

CMD ["nginx", "-g", "daemon off;"]
